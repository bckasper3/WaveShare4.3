# Host (PC) unit tests for envlogger_core. Build + run: powershell -File tools\host_test.ps1
# Every tests/test_<group>.c must define `void run_<group>_tests(void)`; it is found and
# registered automatically (test_registry.h is generated below), so no shared file is edited
# to add tests.
cmake_minimum_required(VERSION 3.16)
project(envlogger_host_tests C)

add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../components/envlogger_core envlogger_core)

file(GLOB TEST_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/tests/test_*.c)
list(SORT TEST_SRCS)
set(DECLS "")
set(CALLS "")
foreach(f ${TEST_SRCS})
    get_filename_component(stem ${f} NAME_WE)
    string(REGEX REPLACE "^test_" "" group ${stem})
    string(APPEND DECLS "void run_${group}_tests(void);\n")
    string(APPEND CALLS "    run_${group}_tests();\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/test_registry.h CONTENT
"/* Generated by test/host/CMakeLists.txt - do not edit. */\n${DECLS}\nstatic void run_all_test_groups(void)\n{\n${CALLS}}\n")

add_executable(host_tests unity/unity.c test_main.c ${TEST_SRCS})
target_include_directories(host_tests PRIVATE unity tests ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(host_tests PRIVATE UNITY_INCLUDE_DOUBLE UNITY_SUPPORT_64)
target_link_libraries(host_tests PRIVATE envlogger_core)

enable_testing()
add_test(NAME host_tests COMMAND host_tests)
